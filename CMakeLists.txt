cmake_minimum_required(VERSION 3.16)
project(crashpad C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CRASHPAD_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
find_package(Threads REQUIRED)
add_library(crashpad src/blockdev.c src/page.c src/pager.c src/store.c src/coordinator.c)
target_include_directories(crashpad PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(crashpad PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(crashpad PRIVATE /W4)
else()
  target_compile_options(crashpad PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(CRASHPAD_SANITIZERS)
    target_compile_options(crashpad PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(crashpad PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(crashpad-shell src/main.c)
target_link_libraries(crashpad-shell PRIVATE crashpad)
add_executable(crashpad-explore tools/explore.c)
target_link_libraries(crashpad-explore PRIVATE crashpad)
add_executable(crashpad-bench tools/bench.c)
target_link_libraries(crashpad-bench PRIVATE crashpad)
if(NOT MSVC)
  target_link_libraries(crashpad-bench PRIVATE m)
endif()
add_executable(crashpad-page-fuzz tests/fuzz_page.c)
target_compile_definitions(crashpad-page-fuzz PRIVATE CRASHPAD_STANDALONE_FUZZ=1)
target_link_libraries(crashpad-page-fuzz PRIVATE crashpad)
include(CTest)
if(BUILD_TESTING)
  add_executable(crashpad-tests tests/test_crashpad.c)
  target_link_libraries(crashpad-tests PRIVATE crashpad)
  add_test(NAME crashpad-tests COMMAND crashpad-tests)
  add_test(NAME crashpad-page-fuzz COMMAND crashpad-page-fuzz)
endif()
